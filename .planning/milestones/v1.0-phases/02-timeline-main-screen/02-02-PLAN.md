    **1. Create apps/web/src/hooks/usePlants.ts:**

    A custom hook that fetches plant data from the API with 60-second polling and falls back to mock data when the server is not available.

    ```typescript
    import { useState, useEffect, useCallback, useRef } from 'react';
    import type { PlantResponse, ISODateString } from '@laplante/shared';
    import { todayISO, addCalendarDays } from '@laplante/shared';
    import { fetchPlants } from '../lib/api';
    import { getMockPlants } from '../lib/mock-data';

    const POLL_INTERVAL_MS = 60_000;
    const PAST_DAYS = 7;
    const FUTURE_DAYS = 7;

    function getDateWindow() {
      const today = todayISO();
      return {
        from: addCalendarDays(today, -PAST_DAYS),
        to: addCalendarDays(today, FUTURE_DAYS),
      };
    }

    export function usePlants() {
      const [plants, setPlants] = useState<PlantResponse[]>([]);
      const [loading, setLoading] = useState(true);
      const [error, setError] = useState<string | null>(null);
      const lastDateRef = useRef<string>(todayISO());

      const fetchPlantsData = useCallback(async () => {
        try {
          const window = getDateWindow();
          const data = await fetchPlants(window.from, window.to);
          setPlants(data);
          setError(null);
        } catch {
          // On first load failure, fall back to mock data
          // On subsequent failures, keep existing data but set error
          setPlants(prev => prev.length === 0 ? getMockPlants() : prev);
          setError('Could not load plants');
        } finally {
          setLoading(false);
        }
      }, []);

      useEffect(() => {
        fetchPlantsData();

        const interval = setInterval(() => {
          // Check for midnight crossover
          const currentDate = todayISO();
          if (currentDate !== lastDateRef.current) {
            lastDateRef.current = currentDate;
            setLoading(true);
          }
          fetchPlantsData();
        }, POLL_INTERVAL_MS);

        return () => clearInterval(interval);
      }, [fetchPlantsData]);

      return { plants, loading, error, retry: fetchPlantsData };
    }
    ```

    Key behaviors:
    - Initial fetch on mount, then re-fetch every 60 seconds
    - If the API fails on first load, use mock data as fallback so the UI is not empty
    - If the API fails on subsequent polls, keep existing data but set error state
    - Detect midnight crossover: when the date changes between polls, set loading to true to trigger re-centering
    - The retry function re-fetches from the API

    **2. Create apps/web/src/hooks/useCurrentTime.ts:**

    A simple hook that returns the current Date, updating every 60 seconds (aligned with the polling interval, since the time badge only needs minute-level precision).

    ```typescript
    import { useState, useEffect } from 'react';

    export function useCurrentTime(updateIntervalMs = 60_000): Date {
      const [now, setNow] = useState(() => new Date());

      useEffect(() => {
        const id = setInterval(() => setNow(new Date()), updateIntervalMs);
        return () => clearInterval(id);
      }, [updateIntervalMs]);

      return now;
    }
    ```
